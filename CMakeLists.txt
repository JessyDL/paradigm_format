cmake_minimum_required(VERSION 3.20)

project(paradigm_format_generator VERSION 0.1.0 LANGUAGES CXX)

option(ENABLE_TESTS "Enable tests" OFF)
option(PFP_INSTALL_HOOKS "Install the git hooks" ON)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Get the latest Git SHA-1
find_package(Git QUIET)
if(GIT_FOUND)
    execute_process(
        COMMAND ${GIT_EXECUTABLE} rev-parse --short HEAD
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
        RESULT_VARIABLE GIT_RESULT
        OUTPUT_VARIABLE GIT_SHA1
        OUTPUT_STRIP_TRAILING_WHITESPACE
    )
    if(NOT GIT_RESULT EQUAL 0)
        set(GIT_SHA1 "unknown")
    endif()
else()
    set(GIT_SHA1 "unknown")
endif()

if(PFP_INSTALL_HOOKS AND GIT_FOUND)
    set(HOOKS_DIR "${CMAKE_SOURCE_DIR}/scripts/hooks")
    set(GIT_HOOKS_DIR "${CMAKE_SOURCE_DIR}/.git/hooks")

    file(MAKE_DIRECTORY "${GIT_HOOKS_DIR}")

    configure_file(${HOOKS_DIR}/pre-commit ${GIT_HOOKS_DIR}/pre-commit COPYONLY)

    if(NOT WIN32)
        execute_process(COMMAND chmod +x ${GIT_HOOKS_DIR}/pre-commit)
    endif()

    message(STATUS "Git pre-commit hook installed")
endif()

set(PROJECT_FULL_VERSION "${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.${PROJECT_VERSION_PATCH}-${GIT_SHA1}")
file(WRITE "${CMAKE_BINARY_DIR}/VERSION" "${PROJECT_FULL_VERSION}\n")

set(VERSION_HEADER_PATH "${PROJECT_SOURCE_DIR}/inc/pfp/version.hpp")
file(WRITE "${VERSION_HEADER_PATH}" 
    "#pragma once\n"
    "// AUTOGENERATED by the project's CMakeLists.txt, do not edit manually\n"
    "#include <string_view>\n"
    "namespace pfp {\n"
    "static constexpr std::string_view PROJECT_VERSION = \"${PROJECT_FULL_VERSION}\";\n"
    "static constexpr unsigned int PROJECT_VERSION_MAJOR = ${PROJECT_VERSION_MAJOR};\n"
    "static constexpr unsigned int PROJECT_VERSION_MINOR = ${PROJECT_VERSION_MINOR};\n"
    "static constexpr unsigned int PROJECT_VERSION_PATCH = ${PROJECT_VERSION_PATCH};\n"
    "static constexpr std::string_view PROJECT_GIT_SHA1 = \"${GIT_SHA1}\";\n"
    "} // namespace pfp\n"
)


add_library(parser STATIC src/parser.cpp src/lexer.cpp "src/ast.cpp")
target_compile_definitions(parser PRIVATE PFP_EXPORTS)
target_include_directories(parser PUBLIC "${PROJECT_SOURCE_DIR}/inc")

# add the generated version header to the includes and mark it as a generated file
target_sources(parser PRIVATE "${VERSION_HEADER_PATH}")
set_source_files_properties("${VERSION_HEADER_PATH}" PROPERTIES GENERATED TRUE)

set(PUBLIC_HEADER_FILES
    ast
    lexer
    parser
    token
)

set(PRIVATE_HEADER_FILES
    lexers/comment
    lexers/float_number
    lexers/keyword
    lexers/literal
    lexers/number
    lexers/operator
    lexers/preprocessor
    lexers/parametric
    lexers/whitespace
)

list(TRANSFORM PUBLIC_HEADER_FILES PREPEND "${PROJECT_SOURCE_DIR}/inc/pfp/")
list(TRANSFORM PUBLIC_HEADER_FILES APPEND ".hpp")
list(TRANSFORM PRIVATE_HEADER_FILES PREPEND "${PROJECT_SOURCE_DIR}/inc/pfp/")
list(TRANSFORM PRIVATE_HEADER_FILES APPEND ".hpp")

target_sources(parser PUBLIC ${PUBLIC_HEADER_FILES} PRIVATE ${PRIVATE_HEADER_FILES})

# Enable warnings and maximum standard compliance
if (MSVC)
    target_compile_options(parser PRIVATE /W4 /WX)
    target_compile_options(parser PRIVATE /permissive- /Zc:preprocessor /Zc:__cplusplus)
    target_compile_options(parser PRIVATE /MP)
else()
    target_compile_options(parser PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(parser PRIVATE -pedantic-errors)
endif()

set_target_properties(parser PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
)

install(TARGETS parser
        ARCHIVE DESTINATION lib
        COMPONENT Libraries)

install(FILES ${PUBLIC_HEADER_FILES}
        DESTINATION include/pfp
        COMPONENT Headers)


if (ENABLE_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()


set(CPACK_PACKAGE_NAME "paradigm_formatter")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}.${PROJECT_VERSION_PATCH}-${GIT_SHA1}")
set(CPACK_PACKAGE_CONTACT "pfp@jessydelannoit.com")

if(WIN32)
    set(CPACK_GENERATOR "ZIP")
else()
    set(CPACK_GENERATOR "TGZ")
endif()

include(CPack)
